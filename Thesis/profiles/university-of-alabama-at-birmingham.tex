% The Format Manual is the authority for UAB and says so: "This UAB Format
% Manual governs the physical appearance of your final document" and "This UAB
% Format Manual takes precedence over the style guide in the following areas".
% Everything below is quoted from it.

\thesisinstitution{University of Alabama at Birmingham}

% "All pages, including preliminary pages, pages with illustrations, landscape
% pages, and appendices must have a 1 1/2 inch left margin in order to allow
% space for binding." / "Top, bottom, and right margins are 1 inch". Restated on
% p. 2: "All pages must adhere to the prescribed margins (1 1/2 inch left, 1
% inch right, top, and bottom)."
%
% The 1.5in left margin is stated for ALL pages, not for odd pages of a bound
% copy, so it is a plain `left' key and not a twoside binding offset -- UAB
% files electronically ("Paper submission is no longer an option") and the
% asymmetry is baked into the filed PDF on every page.
\thesissetgeometry{left=1.5in, right=1in, top=1in, bottom=1in}

% "Use standard double spacing throughout the document." The manual lists three
% exceptions -- long quotes, references, and headings/subheadings/captions --
% which are per-element choices, not a document spacing rule, so the document
% setting stays double.
\thesissetspacing{double}

% THE 2-INCH OPENING IS REQUIRED FOR FRONT MATTER AND OPTIONAL FOR CHAPTERS, and
% this one call covers both because the class applies it to \chapter and
% \chapter* alike.
%
% Required (indicative mood, one per preliminary page): Dedication, "title is
% centered 2 inches from the top of the page"; Acknowledgments, same; Table of
% Contents, "title is centered 2 inches from the top on the first page - 1 inch
% down on the continuing pages"; List of Tables, List of Figures, List of
% Abbreviations, same; Reference Material, "Title is centered 2 inches from top
% on the first page".
%
% Optional for body chapters: the Margins section lists as an exception "(b) the
% first page of each chapter may have a 2-inch top margin", and the Word
% instructions read "You may find it easier to create the 2-inch top margin on
% the first pages of chapters etc." MAY, not must. This profile takes the
% 2-inch option for chapters so that the whole document is consistent with the
% front matter UAB does require it on; a 1-inch chapter opening is equally in
% spec and is obtained by deleting this line.
\thesissetchapteropening{2in}

% The Abstract is the one page the manual holds at 1 inch -- "(This page should
% appear with a one-inch top margin instead of a two-inch top margin)" -- and it
% is not a \chapter in this class, so it is unaffected by the line above. Anyone
% who typesets the abstract as a starred chapter must undo the opening for it.

% Font: "Use 12-point Times New Roman or other common serif type font for all
% text (including headings) throughout the document. (LaTeX users may use the
% default font)". The parenthesis is UAB's, and it is why this profile sets no
% font: the class's Latin Modern is explicitly permitted.

% --- Accessibility -----------------------------------------------------------
% Read on the access date: the Format Manual (all 31 pages, searched for
% "accessib", "alt text", "tagged", "WCAG", "PDF/A", "screen reader", "508" --
% zero hits), the thesis/dissertation Resources index, and the Editing &
% Publishing page. None of them asks for anything of the PDF beyond "a single
% PDF" submitted to ProQuest.
%
% This is a PUBLISHED-NOTHING finding, not an unread section. No
% \thesisrequiretagging here: UAB has not asked for a tagged PDF, and declaring
% that it did would put words in the Graduate School's mouth.
\thesisaccessibilityrequirement{Not published}
	{UAB's Format Manual and the Graduate School's thesis/dissertation resource
	 pages state no accessibility requirement for the filed PDF as of the access
	 date. The class emits a tagged PDF/UA-2 document regardless; nothing here
	 is a UAB rule.}

% --- Title page --------------------------------------------------------------
% The manual prints an annotated specimen (Appendix A, "Sample Title Page") and
% the annotations are prescriptive. The prose adds: "Page is counted as page i
% but not visibly numbered; title is centered between the margins, 2 inches from
% the top of the page and in all capital letters; year of degree conferral is at
% the 1-inch bottom margin."
%
% The specimen's own annotations, in order down the page:
%   "(2-inch top margin)"
%   title -- "(Centered between required margins)", all capitals, single spaced
%   "by"  -- "(Lower case letters approximately 6-8 single spaces below the
%            title)"
%   "(1 regular double space)"
%   author -- "(All capital letters; use the same form of your name for all
%            publications)"
%   "(2 double spaces)"
%   committee -- "(All capital letters; single spaced; do not list degrees;
%            committee chair first, list others in alphabetical order by their
%            last name)"
%   "A DISSERTATION [or THESIS]"
%   "(1 regular double space)"
%   "Submitted to the graduate faculty of The University of Alabama at
%    Birmingham, / in partial fulfillment of the requirements for the degree of"
%   degree -- "Doctor of Philosophy [or Master of Science, etc.]"
%   "(1 regular double space)"
%   "BIRMINGHAM, ALABAMA"
%   "(1 regular double space)"
%   year -- "(Year of graduation on the 1-inch bottom margin)"
%   "(No page number is printed on this page)"
%
% The 2-inch figure is a distance from the PAGE EDGE, so the \vspace* is 1in on
% top of the geometry's own 1in.
%
% \thesis@date is emitted where the specimen puts the year. UAB wants the year
% of degree conferral alone, so a document filing here should write
% \graddate{2027}, not \graddate{May, 2027}. The profile cannot enforce that
% without discarding a value the class lets the author set.
%
% The committee list is redefined locally: UAB collects no signatures inside the
% document, so the class's default signature rule is wrong on this page. The
% chair carries the role text ("JOHN W. SMITH, COMMITTEE CHAIR"); members
% declared with an empty role print the name alone, which is what "do not list
% degrees" and the alphabetical-order rule produce.
\renewcommand{\thesistitlepage}{%
	\loadgeometry{nopagenumber}%
	\begin{titlepage}%
		\thispagestyle{empty}\singlespacing\centering
		\vspace*{1in}%
		\thesis@tagtitle{Title}{\MakeUppercase{\@title}}\par
		\vspace{7\baselineskip}
		by\par
		\vspace{\baselineskip}
		\MakeUppercase{\@author}\par
		\vspace{2\baselineskip}
		\begingroup
		\renewcommand{\thesis@memberline}[2]{%
			\MakeUppercase{##1}\ifx\relax##2\relax\else, \MakeUppercase{##2}\fi\par
		}%
		\thesis@committee
		\endgroup
		\vfill
		\MakeUppercase{A \thesis@docnoun}\par
		\vspace{\baselineskip}
		Submitted to the graduate faculty of The University of Alabama at Birmingham,\\
		in partial fulfillment of the requirements for the degree of\\
		\thesis@degree\par
		\vspace{\baselineskip}
		BIRMINGHAM, ALABAMA\par
		\vspace{\baselineskip}
		\thesis@date\par
	\end{titlepage}%
	\loadgeometry{normal}%
}

% --- Committee approval page --------------------------------------------------
% UAB's filed document contains none.
%
% The manual's preliminary-page list is closed -- "The preliminary pages must be
% arranged in the order listed here and must adhere to these instructions" --
% and its nine entries are Title, Copyright, Abstract, Dedication,
% Acknowledgments, Table of Contents, List of Tables, List of Figures, List of
% Abbreviations. No approval or signature page appears in it, and the sample
% pages in Appendix A show none.
%
% Approval is collected on a separate instrument the Graduate School issues, not
% on a leaf of the manuscript: "Your official Approval Form will be generated
% the day of your defense and sent to your committee electronically", and
% "Signatures of committee members and program directors on Approval Forms
% indicate their assurance that they have examined your document". So signatures
% are still collected, electronically, and not in the PDF that is filed.
\thesisnoapprovalpage
