% REQUIREMENTS THIS CLASS CANNOT ENCODE, all stated in Appendix C:
%   * "Every page of a thesis is assigned a number, although a number may not

%     rest Arabic. Page-number placement is not a profile field.
%   * Page numbers in the lists of figures and tables and in the table of

%   * Table numbers go above the table, figure numbers below the figure, and

\thesisinstitution{Albany College of Pharmacy and Health Sciences}

% "All margins must be one inch except the left, which must be 1.5 inches to
% allow for binding."
\thesissetgeometry{left=1.5in, right=1in, top=1in, bottom=1in}

% "The text of a thesis must be double spaced throughout except for footnotes,
% references, and extended quotations, which are single spaced." Those three
% exceptions are local choices in the document, not a document-wide setting.
\thesissetspacing{double}

% --- Accessibility -----------------------------------------------------------
% Appendix C and the rest of the 48-page handbook were searched for "accessib",
% "WCAG", "tagged", "alt text" and "screen reader": no match.
\thesisaccessibilityrequirement{Not published}
	{ACPHS publishes no accessibility requirement for a filed thesis. The
	 handbook's formatting appendix does not mention accessible or tagged PDFs,
	 alt text, document language or any PDF standard.}
% Recorded separately because it is the one ACPHS rule with an accessibility
% EFFECT while having a different stated purpose. It is quoted rather than
% paraphrased so nobody upgrades it into a WCAG claim ACPHS never made.
\thesisaccessibilityrequirement{Colour is not the only channel}
	{"Black and white figures are preferred. Colored figures must be of
	 sufficient contrast such that information conveyed by different colors can
	 be distinguished in black and white copies." ACPHS gives this for the sake
	 of photocopies, not screen readers — but a figure that satisfies it also
	 satisfies the do-not-encode-by-colour-alone rule other schools state as an
	 accessibility requirement.}

% --- Title page, which is also the approval page ------------------------------
% Reproduced from "Sample Thesis Title Page" in Appendix C. Its prose:
%
%   "The title page includes the thesis title, author, date, submission phrase
%    (submitted to ACPHS faculty in partial fulfillment of the degree of Master
%    of Science in...) and printed names and signatures... The date is provided
%    in month and year format and is the date on which final approval of the
%    thesis or dissertation is granted by the committee. The printed names and
%    signatures are those of the advisor and committee members. Signatures must
%    be done in blue or black ink ONLY."
%
% So this page is signed in wet ink, and the signature grid is four columns --
% Printed Name, Signature, Date, role -- one row per committee member.
%
% The grid is built from \makebox rules rather than a tabular on purpose: it is
% a layout arrangement of blank lines, not tabular data, and tagging it as a
% table would tell a screen reader to announce rows and columns of nothing.
\newcommand{\thesis@acphs@sigrow}{%
	\rule{1.5in}{0.4pt}\hspace{0.15in}%
	\rule{1.5in}{0.4pt}\hspace{0.15in}%
	\rule{0.8in}{0.4pt}\hspace{0.15in}%
	\rule{1.5in}{0.4pt}\par
	\nobreak
	\makebox[1.5in][l]{\small Printed Name}\hspace{0.15in}%
	\makebox[1.5in][l]{\small Signature}\hspace{0.15in}%
	\makebox[0.8in][l]{\small Date}\hspace{0.15in}%
	\makebox[1.5in][l]{\small\thesis@acphs@role}\par
	\vspace{1.5\baselineskip}%
}
\newcommand{\thesis@acphs@role}{}

\renewcommand{\thesistitlepage}{%
	\loadgeometry{nopagenumber}%
	\begin{titlepage}%
		\thispagestyle{empty}\singlespacing\centering
		\vspace*{\fill}
		% The specimen runs the title and the author together -- "<title> by
		% <author>" -- as one centred block.
		\thesis@tagtitle{Title}{\@title{} by \@author}\par
		\vspace{2\baselineskip}
		Submitted to the faculty of the Albany College of Pharmacy and Health
		Sciences in partial fulfillment of the Degree of \thesis@degree{} in
		\thesis@program\par
		\vspace{2\baselineskip}
		Date: \rule{2in}{0.4pt}\par
		\vspace{\baselineskip}
		Approved By: \rule{2in}{0.4pt}\par
		\vspace{2\baselineskip}
		% One signature row per committee member, the roles taken from the
		% document's \committeemember declarations.
		\begingroup
		\raggedright
		\renewcommand{\thesis@memberline}[2]{%
			\renewcommand{\thesis@acphs@role}{##2}%
			\thesis@acphs@sigrow
		}%
		\thesis@committee
		\endgroup
		\vspace*{\fill}
	\end{titlepage}%
	\loadgeometry{normal}%
}
