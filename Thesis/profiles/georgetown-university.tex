% FROM THE TEMPLATE, NOT FROM THE REQUIREMENTS. Every section of
% "Thesis Dissertation Template-PC-OCT 2025.docx" carries
% <w:pgMar w:top="1440" w:right="1440" w:bottom="1440" w:left="1440"/> —
% 1440 twips is exactly one inch — on <w:pgSz w:w="12240" w:h="15840"/>, i.e.
% letter paper. So the template is a 1in square frame with no binding offset.
%
% These are also the class's neutral defaults, so this line changes nothing
% today. It is here because it records the finding and because it is the line
% whoever reads the Box document will edit.
%
% A CAUTION FOR THAT READER. Georgetown pages elsewhere on the web state 1.25in
% for a thesis; the one that does so is english.georgetown.edu's UNDERGRADUATE
% honors thesis page, which is both a department page (not a source under
% RESEARCHING.md) and about a different document. It is named here only so the
% next pass does not treat it as a contradiction it must resolve — but if the
% prose requirements do say 1.25in, this line is wrong and must change.
\thesissetgeometry{left=1in, right=1in, top=1in, bottom=1in}

% FROM THE TEMPLATE: its body paragraphs carry
% <w:spacing w:line="480" w:lineRule="auto"/>, and 480 twentieths of a point
% against a 240 single line is exactly double spacing. The template's abstract
% placeholder says the same in prose — "the text of the abstract begins here and
% continues, double-spaced".
\thesissetspacing{double}

% --- Profile-local metadata ---------------------------------------------------
% The template builds the title page out of dropdowns, so its permitted values
% are known exactly. They are listed here because they are short and because
% guessing one is how a page gets rejected.
%
%   School  — "College of Arts and Sciences", "School of Foreign Service",
%             "School of Health", "School of Medicine"
%   Degree  — "Doctor of Philosophy", "Master of Arts", "Master of Science"
%   Field   — a long controlled list, each entry already carrying its preposition
%             ("in Applied Mathematics", "in Computational Linguistics", …), so
%             the page prints "degree of / <Degree> / in <Field>"
\newcommand{\thesis@gt@school}{}
\newcommand{\georgetownschool}[1]{\renewcommand{\thesis@gt@school}{#1}}
% The author line on the title page is "<full name>, <abbreviation of your
% single highest prior degree earned (with punctuation)>" — e.g. "B.A." or
% "M.S." Optional in the sense that the class cannot know it; Georgetown's
% template shows it filled.
\newcommand{\thesis@gt@prior}{}
\newcommand{\georgetownpriordegree}[1]{\renewcommand{\thesis@gt@prior}{#1}}

% --- Title page --------------------------------------------------------------
% Reproduced from the template's own page, placeholder by placeholder:
%
%   "Type the Title of Your Thesis/Dissertation Here in All Caps: It Should
%    APPear at the top of the page on as few lines as possible"
%   A <Thesis|Dissertation>
%   submitted to the Faculty of the
%   <School>
%   of Georgetown University
%   in partial fulfillment of the requirements for the
%   degree of
%   <Degree written out>
%   <in Field>
%   By
%   "Your First, Middle and Last Names (as they appear on the Personal
%    Information card in GUExperience), Abbreviation of your single highest
%    prior degree earned (with punctuation)"
%   Washington, D.C.
%   Month Day, Year
%
% The title sits AT THE TOP of the page rather than floating, which is why the
% page opens with no leading fill.
%
% \thesis@program supplies the field WITHOUT its preposition — the page adds
% "in" — because that is how every other profile in this directory uses it.
\renewcommand{\thesistitlepage}{%
	\loadgeometry{nopagenumber}%
	\ifx\thesis@gt@school\@empty
		\ClassWarningNoLine{texlib-thesis}{Georgetown's title page names the
			School or College^^J%
			(College of Arts and Sciences, School of Foreign Service, School^^J%
			of Health, or School of Medicine). Set \string\georgetownschool{}}%
	\fi
	\begin{titlepage}%
		\thispagestyle{empty}\singlespacing\centering
		\thesis@tagtitle{Title}{\MakeUppercase{\@title}}\par
		\vspace{4\baselineskip}
		A \thesis@DocNoun\\
		submitted to the Faculty of the\\
		\thesis@gt@school\\
		of Georgetown University\\
		in partial fulfillment of the requirements for the\\
		degree of\\
		\thesis@degree\\
		in \thesis@program\par
		\vspace{3\baselineskip}
		By\par
		\vspace{3\baselineskip}
		\@author\ifx\thesis@gt@prior\@empty\else, \thesis@gt@prior\fi\par
		\vspace*{\fill}
		Washington, D.C.\\
		\thesis@date\par
		\vspace{2\baselineskip}
	\end{titlepage}%
	\loadgeometry{normal}%
}
