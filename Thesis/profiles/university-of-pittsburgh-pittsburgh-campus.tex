\thesisinstitution{University of Pittsburgh}

% "The top, bottom, left, and right margins should be 1". Tables and figures use
% the same margins as text. The student's school or department must give written
% approval for exceptions to margin size."
\thesissetgeometry{left=1in, right=1in, top=1in, bottom=1in}

% "The text of the document may be double-spaced or one-and-one-half spaced with
% the exception of long quotations, footnotes, bibliographical references, and
% the index, which may be single-spaced."
%
% Pitt permits either, so double is a choice within the rule rather than the
% rule itself; \thesissetspacing{onehalf} is equally in spec.
\thesissetspacing{double}

% --- Accessibility -----------------------------------------------------------
% Pitt states accessibility as a first-order concern of the manual rather than
% as an appendix, and publishes a checklist for it.
\thesisrequiretagging
\thesisaccessibilityrequirement{Stated as crucial}
	{"Of crucial importance to each thesis or dissertation are the issues of
	 legibility and accessibility. It is important that ETDs have high-quality
	 illustrations that are clear and contain relevant alt-text descriptions."}
\thesisaccessibilityrequirement{Alt text}
	{Illustrations must carry relevant alt-text descriptions.}
\thesisaccessibilityrequirement{ETD Accessibility Checklist}
	{The ETD guidelines publish a separate accessibility checklist; work through
	 it before submitting.}
\thesisaccessibilityrequirement{Typeface}
	{"Text must be set in 10 points or larger type size and of a fully legible
	 font. Script fonts are not acceptable." Students are advised to confirm
	 with their school that files "will meet University standards of legibility
	 and accessibility".}
