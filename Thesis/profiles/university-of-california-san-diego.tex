% VERIFIED and set here: geometry, spacing.
% LEFT TO THE NEUTRAL FALLBACK: title page and approval page. UCSD DOES require
% an approval page -- the manual carries four specimen layouts of it, one per
% degree path (PhD/DMA, joint EdD with CSUSM, joint PhD with SDSU, and the
% master's) -- so the neutral page stands in rather than a guess at which. The
% class's fallback is a plain approval leaf, which is the safer wrong answer
% than a confidently mislabelled one.

\thesisinstitution{University of California San Diego}

% "Minimum Margins ... TOP LINE OF TYPE = 1" from top of paper; BOTTOM LINE OF
% TYPE = 1" from bottom of paper; LEFT = 1" from left side of paper; RIGHT = 1"
% from right side of paper." The manual adds that "Students may set slightly
% larger margins but must be sure that the final text lies well within these
% guidelines" -- these are minima, so the class's exact 1in conforms.
\thesissetgeometry{left=1in, right=1in, top=1in, bottom=1in}

% "All text must be double spaced, except as indicated below." (The exceptions
% are the usual long quotations, notes and table matter.)
\thesissetspacing{double}

% --- Accessibility -----------------------------------------------------------
% The manual frames its purpose in terms of access -- "to make it as accessible
% as possible to future scholars and researchers who seek to consult it" -- but
% that is about archival availability, not assistive technology, and it is not
% recorded here as though it were a digital-accessibility rule.
\thesisaccessibilityrequirement{Not published}
	{The formatting manual states no digital-accessibility requirement for the
	 filed manuscript.}
\thesisaccessibilityrequirement{Typeface}
	{A font "that is highly legible, dark enough to reproduce clearly, and large
	 enough to be readily" read.}
