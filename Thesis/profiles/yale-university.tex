\thesisinstitution{Yale University}

% "1 inch margin on all sides", with page numbers "0.5 inch from any edge".
%
% Yale's binding exception is conditional on the student's own choice: "If you
% will be ordering softbound copies the left margin should be 1.5 inches." The
% filed PDF is not the softbound copy, so 1in is the default here. A candidate
% who is ordering bound copies should say
%     \thesissetgeometry{left=1.5in, right=1in, top=1in, bottom=1in}
% after loading this profile.
\thesissetgeometry{left=1in, right=1in, top=1in, bottom=1in}

% "Double-spaced", with the bibliography "single-spaced, with a double-space
% between each entry".
\thesissetspacing{double}

% --- Accessibility -----------------------------------------------------------
% One rule here is unusually strict and is worth stating even though it is a
% typographic rule rather than an accessibility one: the font must be uniform
% across furniture as well as text.
\thesisaccessibilityrequirement{Typeface}
	{"10 to 12-point font", and the "same font type should be used throughout,
	 including header, footnotes, page numbers".}
\thesisaccessibilityrequirement{Not published}
	{The 2025 formatting document states no accessibility requirement for the
	 filed dissertation -- no tagging, alt-text or WCAG obligation is given.}
