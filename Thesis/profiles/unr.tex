% profiles/unr.tex — University of Nevada, Reno
% ============================================================================
% The reference institution profile for texlib-thesis.cls, and the one to copy
% when writing another. It reproduces exactly what thesis.cls emitted before the
% class was split, so `\documentclass{thesis}' (the wrapper, which passes
% profile=unr) is unchanged for every existing document.
%
% A profile owns three things and nothing else:
%   * \thesisinstitution — the name used in warnings and on the pages
%   * \thesistitlepage / \thesisapprovalpage — the two page layouts
%   * optionally \thesissetgeometry / \thesissetspacing — the filing figures
%
% Loaded LAST by the class, so everything it needs (fontspec faces, graphicx,
% the metadata macros) already exists -- and, being still in the preamble, a
% profile may \RequirePackage anything else it needs.
%
% tikz is the one thing this profile has to ask for. The class stopped loading
% it by default because this is the ONLY profile of twenty-one that draws, and
% the other twenty were paying ~0.34s a pass for a frame they never render.
%
% PROTOTYPE, and the wording is the part that rots: check the committee-page
% text and the margin rule against the current Graduate School filing
% guidelines before a real submission. Built on Paul Hurtado's accessible
% template (v0.5, pauljhurtado.com/latex).

% Idempotent, and it brings tikzpagenodes with it -- that is what supplies
% `current page text area' in \thesisapprovalpage below.
\TeXLibLoadTikz

\thesisinstitution{University of Nevada, Reno}

% UNR requires 1.0in on ALL FOUR sides, with no exception for binding -- tables
% and appendices included. The manuscript is filed electronically ("no hard
% copies will be produced"), so the wider inner margin a bound copy would want
% is not something the filed document may have: an inner=1.25in twoside build is
% 0.25in out of spec on every odd page and would be caught by the Graduate
% School's margin check. twoside therefore stays a LAYOUT option only -- it still
% governs where headers and \cleardoublepage blanks fall -- and does not change
% the margins. Add a binding offset at print time, not in the filed PDF.
%
% These are the class defaults too, so the call is redundant today. It is here
% anyway because it is the line a new profile edits, and because a later change
% to the neutral default must not silently move UNR's filed margins.
\thesissetgeometry{left=1in, right=1in, top=1in, bottom=1in}
\thesissetspacing{double}

% TeX Gyre Termes gives the committee page its required Times-like face. Latin
% Modern (the class default elsewhere) satisfies UNR's "Times New Roman, Arial,
% or a similarly clear font" and embeds every face, which the Graduate School
% requires.
\newfontfamily{\termesfont}{TeX Gyre Termes}

% --- Title page --------------------------------------------------------------
% UNR: unnumbered, centred. The title is emitted through \thesis@tagtitle so a
% future explicit \tagstructbegin can wrap it without touching this file.
\renewcommand{\thesistitlepage}{%
	\loadgeometry{nopagenumber}%
	\begin{titlepage}%
		\thispagestyle{empty}\singlespacing\centering
		\vspace*{\fill}
		\textsc{University of Nevada, Reno}\par
		\vfill\vfill\vfill
		\thesis@tagtitle{Title}{{\large\bfseries\@title}}\par
		\vfill\vfill
		A \thesis@docnoun{} submitted in partial fulfillment of the\\
		requirements for the degree of \thesis@degree{} in\\
		\thesis@program\par
		\vfill\vfill
		by\\[1em]{\bfseries\@author}\par
		\vfill\vfill
		\thesis@advisor~/ \thesis@DocNoun{} Advisor\par
		\vfill
		\thesis@date\par
		\vspace*{\fill}
	\end{titlepage}%
	\loadgeometry{normal}%
}

% --- Committee approval page --------------------------------------------------
% Sources, read 2026-10-30:
%   FG  https://www.unr.edu/grad/current-students/filing-guidelines
%       section 4 (instructions for the page) and section 5 (Committee approval
%       page; Format and sequence of pages)
%   S   the sample committee pages linked from FG section 6 (thesis and
%       doctoral, both dated August 2023)
% FG section 6 also links Word templates for 3-6 member committees, with and
% without a co-advisor; they sit behind a Box login and were NOT read.
%
% What every reading agrees on: the Graduate School heading, "We recommend that
% the thesis/dissertation" / "prepared under our supervision by", "entitled" in
% lowercase, the title in bold with no hyphenation, "be accepted in partial
% fulfillment of the" / "requirements for the degree of", the degree without the
% program, one entry per member with the Graduate School Representative last,
% then the Dean of the Graduate School ("The last entry is reserved for the Dean
% of the Graduate School", FG 4), then the month and year. No page number and
% "NO SIGNATURES" (FG 3, 5) -- so no signature rules either, which this page used
% to draw.
%
% Where FG and S disagree, the default follows FG's prose and a key selects the
% other reading (see \thesisset in the class):
%
%   committee-page   = typeset | pdf     FG 4 says to type the words as they
%       appear on the template; FG 5 says to use the PDF template and "Combine
%       the PDF into your manuscript". `pdf' places committee-pdf=<file> (the
%       filled-in template, exported to PDF) as a full page, as an image with
%       alt text: its own text and tags do not survive, which is why `typeset'
%       is the default.
%   committee-roles  = below | inline    FG 4 puts each member's role "under
%       each one", and S sets it on its own italic line; FG 4's advisor example
%       is one line, "Sonia A. Skakich, Ph.D., Advisor".
%   committee-author = upper | as-written   FG 4: "ALL CAPITAL LETTERS and
%       BOLD-FACED"; S prints the name in bold, mixed case.
%   committee-degree = upper | as-written   FG 4: "all CAPITAL LETTERS and
%       Bold-Faced"; S prints "Doctor of Philosophy" in bold, mixed case.
%
% A double blue frame as in S. The hidden tagged "Committee Approval Page" H1
% gives a screen reader a heading; the frame is decorative and drawn with a tikz
% overlay so it contributes no structure.
\newif\ifthesis@unr@pdfpage
\newif\ifthesis@unr@rolesbelow   \thesis@unr@rolesbelowtrue
\newif\ifthesis@unr@authorupper  \thesis@unr@authoruppertrue
\newif\ifthesis@unr@degreeupper  \thesis@unr@degreeuppertrue
\newcommand{\thesis@unr@pdf}{}
\ExplSyntaxOn
\keys_define:nn { thesis }
	{
		committee-page .choices:nn = { typeset, pdf }
			{ \use:c { thesis@unr@pdfpage \int_compare:nNnTF \l_keys_choice_int = 2 { true } { false } } } ,
		committee-pdf .tl_set:N = \thesis@unr@pdf ,
		committee-roles .choices:nn = { below, inline }
			{ \use:c { thesis@unr@rolesbelow \int_compare:nNnTF \l_keys_choice_int = 1 { true } { false } } } ,
		committee-author .choices:nn = { upper, as-written }
			{ \use:c { thesis@unr@authorupper \int_compare:nNnTF \l_keys_choice_int = 1 { true } { false } } } ,
		committee-degree .choices:nn = { upper, as-written }
			{ \use:c { thesis@unr@degreeupper \int_compare:nNnTF \l_keys_choice_int = 1 { true } { false } } } ,
	}
\ExplSyntaxOff

\newcommand{\thesis@unr@upper}[2]{% #1 = the if-name's test, #2 = text
	#1{\MakeUppercase{#2}}\else{#2}\fi}

% One entry: name and degree, then the role -- on its own italic line (S) or
% after a comma (FG 4's advisor example). Used for the Dean too.
\newcommand{\thesis@unr@memberline}[2]{%
	\par\vskip 0.45\baselineskip
	\ifthesis@unr@rolesbelow
		#1\par
		\ifx\relax#2\relax\else{\itshape #2}\par\fi
	\else
		#1\ifx\relax#2\relax\else, #2\fi\par
	\fi
}

\newsavebox{\thesis@unr@box}
\newcommand{\thesis@unr@typeset}{%
	\clearpage\thispagestyle{empty}%
	\loadgeometry{nopagenumber}\singlespacing
	% Built in a box first, so a page that is too full -- a long title and a
	% six-member committee with a co-advisor -- is reported here instead of
	% spilling its last lines onto a numbered page of their own. The box is used
	% exactly once, which is what keeps its tagged content in the structure tree.
	% The frame is inside it so that the box is the page's only item.
	\setbox\thesis@unr@box=\vbox{\hsize=\textwidth
	\noindent\begin{tikzpicture}[remember picture, overlay]
		\draw[draw={rgb,255:red,0;green,0;blue,128}, line width=0.75pt]
			([xshift=1.5pt,yshift=-1.5pt]current page text area.north west)
			rectangle
			([xshift=-1.5pt,yshift=1.5pt]current page text area.south east);
		\draw[draw={rgb,255:red,0;green,0;blue,128}, line width=0.75pt]
			([xshift=3pt,yshift=-3pt]current page text area.north west)
			rectangle
			([xshift=-3pt,yshift=3pt]current page text area.south east);
	\end{tikzpicture}\par
	\begin{center}
		\thesis@tagtitle{H1}{{\color{white}\tiny Committee Approval Page}}\par
		\vspace{-1em}
		\includegraphics[width=0.68in, alt={University of Nevada, Reno logo}]{UNRlogoN}\par
		\termesfont
		\vspace{1mm}
		{\fontsize{12}{18}\selectfont\textcolor[RGB]{0,0,128}{THE GRADUATE SCHOOL}}\par
		\vfill
		{\fontsize{12}{12}\selectfont We recommend that the \thesis@docnoun\\
			prepared under our supervision by}\par
		\vfill
		{\bfseries\thesis@unr@upper\ifthesis@unr@authorupper{\@author}}\par
		\vfill
		{\fontsize{12}{12}\selectfont entitled}\par
		% [t]: a centered minipage straddles the line before it, so a title of
		% two or more lines printed on top of "entitled".
		\begin{minipage}[t]{0.7\textwidth}
			\centering\bfseries
			\hyphenpenalty=10000 \exhyphenpenalty=10000 \emergencystretch=2em
			\@title\par
		\end{minipage}\par
		\vfill
		{\fontsize{12}{12}\selectfont be accepted in partial fulfillment of the\\
			requirements for the degree of}\par
		\vfill
		{\bfseries\thesis@unr@upper\ifthesis@unr@degreeupper{\thesis@degree}}\par
		\vfill
		\let\thesis@memberline\thesis@unr@memberline
		\thesis@committee
		\ifx\thesis@dean\@empty
			\ClassWarningNoLine{texlib-thesis}{The UNR committee approval page ends with^^J%
				the Dean of the Graduate School, and no \string\graddean\space is set.^^J%
				Add \string\graddean{Name, Ph.D.} with the name on the Graduate^^J%
				School's current template}%
		\else
			\thesis@memberline{\thesis@dean, Dean}{Graduate School}%
		\fi
		\vfill
		\thesis@date\par
	\end{center}}%
	\@tempdima=\dimexpr\ht\thesis@unr@box+\dp\thesis@unr@box\relax
	\ifdim\@tempdima>\textheight
		\@tempdima=\dimexpr\@tempdima-\textheight\relax
		\ClassWarningNoLine{texlib-thesis}{The committee approval page is %
			\the\@tempdima\space too tall for^^J%
			one page and runs into the bottom margin. Break the title into^^J%
			fewer lines, or set committee-roles=inline}%
	\fi
	\vbox to \textheight{\unvbox\thesis@unr@box}%
	\loadgeometry{normal}\doublespacing\clearpage
}

% committee-page=pdf: the filled-in Graduate School template, as a full page.
\newcommand{\thesis@unr@pdfpage}{%
	\clearpage
	\newgeometry{margin=0pt, nohead, nofoot}%
	\thispagestyle{empty}%
	\begingroup
		% Alt text from the same metadata the typeset page uses, as plain text.
		\def\thesis@memberline##1##2{##1, ##2; }%
		\def\\{ }\let\newline\\%
		\protected@edef\thesis@unr@alt{Committee approval page. The Graduate School. We
			recommend that the \thesis@docnoun\space prepared under our supervision
			by \@author\space entitled \@title\space be accepted in partial
			fulfillment of the requirements for the degree of \thesis@degree.
			\thesis@committee
			\ifx\thesis@dean\@empty\else\thesis@dean, Dean, Graduate School. \fi
			\thesis@date}%
		% Exactly one text height tall: a paragraph line of the same height did
		% not fit, and left a blank page before a numbered one.
		\vbox to \textheight{\vss
			\hbox to \textwidth{\hss\includegraphics[width=\paperwidth,
				height=\paperheight, keepaspectratio,
				alt=\thesis@unr@alt]{\thesis@unr@pdf}\hss}%
			\vss}%
	\endgroup
	\clearpage\loadgeometry{normal}%
}

\renewcommand{\thesisapprovalpage}{%
	\ifthesis@unr@pdfpage
		\ifx\thesis@unr@pdf\@empty
			\ClassWarningNoLine{texlib-thesis}{committee-page=pdf needs
				committee-pdf=<file>.^^J%
				Typesetting the committee approval page instead}%
			\thesis@unr@typeset
		\else
			\thesis@unr@pdfpage
		\fi
	\else
		\thesis@unr@typeset
	\fi
}
