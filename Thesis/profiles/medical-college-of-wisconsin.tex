% --- Accessibility -----------------------------------------------------------
% Searched: the format document, the Degree Completion page, the 2025–2026
% handbook in full, and the Graduate Student Forms page. None of them asks for
% an accessible or tagged PDF, alt text, a document language, or a PDF standard.
\thesisaccessibilityrequirement{Not published}
	{The School of Graduate Studies publishes no accessibility requirement for a
	 filed dissertation. MCW's Web Accessibility Statement commits the
	 \emph{website} to WCAG 2.1 Level AA; it is not addressed to a filed
	 dissertation and is not reproduced as one here.}

% --- Title page --------------------------------------------------------------
% MCW requires a title page — it is first in the prescribed order — but publishes
% no layout or wording for it and no specimen. The class's neutral title page is
% therefore left in place, and a candidate should check the wording with the
% Graduate School before filing.

% --- Committee page ----------------------------------------------------------
% MCW's prescribed order is: "Title page / List of Committee Member Names (no
% signatures) / Abstract / ...". The parenthesis is the operative part: the
% neutral approval page draws a \rule signature line under every member, which
% this school's filed document must not have. So the page is rebuilt as what MCW
% names it — a list of committee member names.
%
% \thesisnoapprovalpage would be wrong here: MCW requires the page, it just
% requires it unsigned.
%
% The heading wording is MCW's own name for the page. Nothing else about the
% layout is published, so the rest is plain and centred.
\renewcommand{\thesisapprovalpage}{%
	\clearpage\thispagestyle{empty}%
	\loadgeometry{nopagenumber}\singlespacing
	\begin{center}
		\thesis@tagtitle{H1}{{\large\bfseries List of Committee Member Names}}\par
		\vspace{2\baselineskip}
		% Names only: no signature rule, no role suppressed -- MCW forbids the
		% signature, not the role.
		\begingroup
		\renewcommand{\thesis@memberline}[2]{%
			##1\ifx\relax##2\relax\else, ##2\fi\par\vspace{0.5\baselineskip}%
		}%
		\thesis@committee
		\endgroup
		\vfill
		\thesis@date\par
	\end{center}%
	\loadgeometry{normal}\doublespacing\clearpage
}
