\thesisinstitution{University of Massachusetts Chan Medical School}

% "Margins must be at least 1 1/2 inches wide at the top and left, and 1-inch
% wide at the bottom and right. Illustrations, tables and figures should conform
% to these margins as well." These are minima; the profile takes the minimum.
\thesissetgeometry{left=1.5in, right=1in, top=1.5in, bottom=1in}

% "All main text should be double spaced (3 lines per vertical inch)". The guide
% permits single spacing for "the table of contents, tables, figure legends,
% long quotations, footnotes, and bibliographical citations" — those are local
% choices in the document, not a document-wide setting.
\thesissetspacing{double}

% The typeface rule is a PREFERENCE, not a requirement -- "The preferred font
% size is 12 in either the Times New Roman or Arial styles" -- so nothing is
% forced here. A candidate who wants to follow it selects the face in their own
% preamble.

% --- Accessibility -----------------------------------------------------------
% Searched the preparation guide and the 64-page student handbook for
% "accessib", "alt text", "tagged", "WCAG", "PDF/A" and "screen reader": no
% match in either. Neither document asks for anything.
\thesisaccessibilityrequirement{Not published}
	{The Morningside Graduate School publishes no accessibility requirement for
	 a filed dissertation in either the preparation guide or the student
	 handbook.}

% The Dean signs off on the reviewer page BY NAME in the guide's specimen. A

% --- Title page --------------------------------------------------------------
% "Sample Title Page – Formatting for the Title Page", reproduced. The guide's
% parenthesised placeholders — (NAME IN FULL), (MONTH, DAY AND YEAR),
% (MAJOR SUBJECT) — mark fields, not literal text, and are filled from the
% document's metadata.
%

	\begin{titlepage}%

		\vspace{\baselineskip}
		Morningside Graduate School of Biomedical Sciences at
		UMass Chan Medical School\par
		\vspace{\baselineskip}
		in partial fulfillment of the requirements for the degree of\par
		\vspace{2\baselineskip}
		\MakeUppercase{\thesis@degree}\par
		\vspace{2\baselineskip}
		\MakeUppercase{\thesis@date}\par
		\vspace{2\baselineskip}
		\MakeUppercase{\thesis@program}\par

	\end{titlepage}%
	\loadgeometry{normal}%
}
