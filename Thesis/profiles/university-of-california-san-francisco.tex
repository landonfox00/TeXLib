\thesisinstitution{University of California, San Francisco}

% "Every page of your thesis or dissertation must have a one-inch margin on all
% sides, top, bottom, left, and right. The only element that may be placed
% outside of the one-inch margin is the page number, which should print no
% closer than one-half inch from the edge of the paper."
%
% The class's `normal' geometry uses includefoot, so the folio sits inside the
% one-inch band rather than out in the margin. That is stricter than UCSF asks
% and therefore in spec; a profile only has to avoid going the other way.
\thesissetgeometry{left=1in, right=1in, top=1in, bottom=1in}

% "Double spacing should be used except in those places where conventional usage
% calls for single spacing such as long quotes, footnotes, indented quotations,
% figures, and tables. Abstract and dedication pages should also be
% double-spaced."
\thesissetspacing{double}

% --- Accessibility -----------------------------------------------------------
% Nothing on the formatting page states an accessibility requirement. The one

% reading it as an accessibility rule, so it is quoted rather than paraphrased.
\thesisaccessibilityrequirement{Not published}
	{The Graduate Division's formatting page states no accessibility
	 requirement for the filed manuscript.}
\thesisaccessibilityrequirement{Colour}
	{"Native digital manuscripts will appear in color when viewed
	 electronically. Microfilm and print reproductions will NOT preserve color."
	 Do not let colour alone carry meaning.}
