\thesisinstitution{Columbia University in the City of New York}

% "All four margins of the page must be at least one inch, and the left and
% right margins should be equal. All material, including figures, tables,
% headers/footers, footnotes/endnotes, and images, must appear within the
% margins of the manuscript. Page numbers are the only exception; these must
% appear at least three quarters of an inch (3/4") from the edge of the page,
% but can appear in the one-inch margin."
%
% The equal-left-and-right rule is the one to watch: it rules out the asymmetric
% binding margin that Duke and Georgia Tech require, so this geometry must not
% be "improved" by copying one of those.
\thesissetgeometry{left=1in, right=1in, top=1in, bottom=1in}

% "The text, abstract, dedication, acknowledgments, and table of contents are
% double spaced. Normally, quotations of eight lines or more are indented and
% single spaced. Footnotes, endnotes, and bibliographic entries should be single
% spaced with double spacing between each entry."
\thesissetspacing{double}
