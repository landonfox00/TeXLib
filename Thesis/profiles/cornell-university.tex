% VERIFIED and set here: geometry, spacing.
% LEFT TO THE NEUTRAL FALLBACK: title page and approval page; the formatting
% requirements page covers page setup, font, margins, spacing, tables and
% figures, and does not describe either page.

\thesisinstitution{Cornell University}

% "Margins should be at least 1 inch with page numbers at least 3/4 inch from
% the edge of the page. The templates use 1.6 inches for the left margin and 1.1
% inches for the right."
%
% The stated RULE is the 1 inch minimum; the 1.6/1.1 figures describe what
% Cornell's own templates happen to do, and the page does not require them. This
% profile encodes the rule, not the template, so a document built here is in
% spec without being locked to a layout Cornell does not mandate. Anyone who
% wants to match the template exactly can say so:
%     \thesissetgeometry{left=1.6in, right=1.1in, top=1in, bottom=1in}
\thesissetgeometry{left=1in, right=1in, top=1in, bottom=1in}

% "Document must be double-spaced with the exception of quotations as
% paragraphs, captions, lists, graphs, charts, footnotes/endnotes, bibliographic
% entries, items within tables, and lists in appendices." Irregular spacing is
% permitted "to accommodate poetry or other creative writing".
\thesissetspacing{double}

% --- Accessibility -----------------------------------------------------------
% Cornell frames its requirements around archival preservation -- they "meet UMI
% standards and American Library Association (ALA) suggestions for preserving
% archival copies" -- rather than around assistive technology. \thesisrequiretagging
% is therefore NOT set: this class produces a tagged PDF anyway, but that is the
% library's doing and Cornell does not ask for it, and a profile that claimed
% otherwise would put words in the Graduate School's mouth.
\thesisaccessibilityrequirement{Embedded text}
	{"Text must be embedded, 11-point or larger font." The requirement applies
	 to all text, including captions, footnotes and citations.}
\thesisaccessibilityrequirement{Colour}
	{"Black text is recommended; although, color may be appropriate in some
	 limited parts of the document."}
\thesisaccessibilityrequirement{Not published}
	{The formatting requirements state no assistive-technology obligation --
	 no tagging, alt-text or WCAG requirement is given.}
