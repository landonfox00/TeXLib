% --- Accessibility -----------------------------------------------------------
% The Manual's rule is one sentence and it is unconditional: "VCU is committed
% to providing equal access to programs, online content, services, and
% activities. All submissions must be accessible." The ETD checklist repeats it
% as a line item — "All documents are accessible".
%
% The detail lives in the accessibility document, which asks the filer to
% "Confirm PDF tagging" and to "Generate tagged PDFs rather than print-to-PDF
% outputs", so \thesisrequiretagging is set.
%
% \thesisrequirepdfstandard is deliberately NOT set, and the reason is the modal
% verb. PDF/A appears only in a list of formats that "are most likely to remain
% viable in the future" — "Other recommended formats include: Text — PDF or
% PDF/A (.pdf)" — with plain PDF named in the same breath. PDF/UA appears only
% as a property of a tool VCU recommends ("PAC ... validates PDFs against PDF/UA
% standards"). Neither is required, and declaring one would put a standard in
% VCU's mouth that it never named.
%
% \thesisrequiredocumentlanguage is NOT set either: no source asks for the
% document language, even though WCAG conformance would imply it. VCU names WCAG
% without naming a version or a level, so the chain is too loose to declare.
%
% VCU's ADA Title II citation carries NO compliance date. Nothing here to
% reconcile against another profile's date.
\thesisrequiretagging
\thesisaccessibilityrequirement{Rule}
	{"All submissions must be accessible." The ETD checklist carries it as a
	 completion item: "All documents are accessible".}
\thesisaccessibilityrequirement{Why}
	{"Public universities are required to ensure that their digital content is
	 accessible under Title II of the Americans with Disabilities Act (ADA).
	 Accessible digital content conforms to the Web Content Accessibility
	 Guidelines (WCAG)". No WCAG version or level is named, and no compliance
	 date is given.}
\thesisaccessibilityrequirement{Tagging}
	{The filer must "Confirm PDF tagging" before submitting, and "Generate
	 tagged PDFs rather than print-to-PDF outputs".}
\thesisaccessibilityrequirement{Headings}
	{Heading styles are to be used "from the start", in a stated hierarchy:
	 Heading 1 for the document title (used once), Heading 2 for chapter titles,
	 Heading 3 for major subsections, Heading 4 for subsections. In LaTeX: "Use
	 semantic structure commands consistently (\textbackslash section,
	 \textbackslash subsection, etc.)".}
\thesisaccessibilityrequirement{Alt text}
	{"All images must include alt-text" (ETD sample template); the accessibility
	 guidance asks for "Meaningful alternative text for figures and diagrams".}
\thesisaccessibilityrequirement{Mathematics}
	{"Equations and formulas should be created using tools that preserve the
	 underlying mathematical structure (such as Word's equation editor or
	 LaTeX). Avoid inserting equations as images whenever possible, as this
	 prevents assistive technologies from interpreting the mathematical
	 content." For LaTeX specifically: "Use accessible math tagging packages
	 (e.g., tagged PDF workflows now available in newer LaTeX distributions)".}
\thesisaccessibilityrequirement{Tables}
	{"Use accessible table formatting"; review "alt text and tables" before
	 submitting.}
\thesisaccessibilityrequirement{Reading order and navigation}
	{"Logical reading order" and "Bookmarks and navigational structure" are
	 listed among the points needing particular attention in a maths-heavy
	 document.}
\thesisaccessibilityrequirement{Checking}
	{"Run accessibility checkers" before submitting. VCU names Adobe Acrobat
	 Pro's Accessibility Checker, PAC (the PDF Accessibility Checker, "a free
	 tool that validates PDFs against PDF/UA standards"), and Word's checker. It
	 also warns that "the Library is not able to provide individual
	 accessibility reviews of dissertations".}

% --- Title page --------------------------------------------------------------
% The Manual states the required CONTENT: "The title page includes the student's
% name, the date the advisory committee approved the thesis/dissertation, and
% the partial fulfillment statement", and gives the statement verbatim:
%
%   "A thesis/dissertation submitted in partial fulfillment of the requirements
%    for the degree of (list degree, for example, Master of Science, Doctor of
%    Philosophy) at Virginia Commonwealth University."
%
% The sample template supplies the ORDER and the remaining lines, and the block
% below is that page: title; the statement; "by"; the student's name with prior
% degrees; the director; the place; the date. The template's own words for the
% name and director lines are "(Student's name in full, Student's degrees with
% Institutions and dates)" and "Director: Thesis/Dissertation director's name,
% Title, and department" — so both slots carry more than the class's metadata
% model holds on its own.
%
% \graddate is the date the committee approved the work, and the template writes
% it as "Month, Year" — which for once is exactly what \graddate defaults to.
%
% \vcupriordegrees{...} is provided for "Student's degrees with Institutions and
% dates", which the class has no field for. It prints on the same line as the
% author's name, as the template does. Empty by default.
%
% No vertical positions are prescribed by any source, so the page is centred
% between the margins rather than pinned to invented offsets.
\newcommand{\thesis@vcu@priordegrees}{}
\newcommand{\vcupriordegrees}[1]{\renewcommand{\thesis@vcu@priordegrees}{#1}}

\renewcommand{\thesistitlepage}{%
	\loadgeometry{nopagenumber}%
	\begin{titlepage}%
		\thispagestyle{empty}\singlespacing\centering
		\vspace*{\fill}
		\thesis@tagtitle{Title}{{\large\bfseries\@title}}\par
		\vspace{2\baselineskip}
		\begin{minipage}{0.85\textwidth}
			\centering
			A \thesis@docnoun{} submitted in partial fulfillment of the
			requirements for the degree of \thesis@degree{} at Virginia
			Commonwealth University.\par
		\end{minipage}\par
		\vspace{2\baselineskip}
		by\par
		\vspace{\baselineskip}
		\@author\ifx\thesis@vcu@priordegrees\@empty\else,
			\thesis@vcu@priordegrees\fi\par
		\vspace{2\baselineskip}
		Director: \thesis@advisor\par
		\vspace{2\baselineskip}
		Virginia Commonwealth University\\
		Richmond, Virginia\par
		\thesis@date\par
		\vspace*{\fill}
	\end{titlepage}%
	\loadgeometry{normal}%
}

%
