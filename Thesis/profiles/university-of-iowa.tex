\thesisinstitution{The University of Iowa}

% "Margins must be a minimum of 1 inch on all sides and on all pages, including
% the Preliminary Pages." / "Margins should be consistent throughout the entire
% text."
%
% A MINIMUM, not a fixed figure -- a wider margin is in spec, provided it is
% consistent. 1in is the smallest compliant value and is what this profile sets.
\thesissetgeometry{left=1in, right=1in, top=1in, bottom=1in}

% "Line spacing may be either 1.5 or double-spaced. Line spacing should be
% consistent throughout."
%
% A CHOICE. This profile takes double, the class default;
% \thesissetspacing{onehalf} is equally in spec.
\thesissetspacing{double}

% No chapter-opening exception: Iowa's Headings requirements are about
% consistency ("Spacing around major headings should be consistent throughout
% the entire manuscript") and name no figure. \thesissetchapteropening is
% deliberately not called.
%
% Type size: "Use 10-12-point font for the body of the manuscript. Font smaller
% than 8-point is not allowed. You may use 12-point font for major headings.
% Font larger than this may be used sparingly, if at all." The class's 12pt body
% is at the top of the permitted range -- note that this makes 12pt the ceiling
% for headings too, so a document filing here should not enlarge them. "A range
% of font styles is acceptable", so no face is set.

% --- Accessibility -----------------------------------------------------------
% Iowa asks for very little, and what it does ask is a CONSIDER, not a
% requirement. The whole 20-page manual has three hits for "accessib" and two of
% them are about open access. \thesisrequiretagging is not set; neither is
% anything else.
\thesisaccessibilityrequirement{Alt text (suggested)}
	{"Consider adding alt text that describes any figures you include. As you
	 craft alt text for your figures, be sure to explain the content and function
	 of each as clearly and accurately as possible. Try to keep the text
	 succinct." Iowa's own verb is "Consider" -- this is guidance, not a filing
	 requirement.}
\thesisaccessibilityrequirement{Nothing else published}
	{No other accessibility requirement for a filed thesis appears in "Thesis and
	 Dissertation at Iowa" (Spring 2025) as of the access date -- no tagged PDF,
	 no PDF standard, no document language, no WCAG or Title II reference. The
	 one PDF-conversion instruction is about bookmarks, not tags: "make sure
	 'Convert Word Headings to Bookmarks' is checked."}

% --- Title page --------------------------------------------------------------
% Appendix A fixes the CONTENT; the Graduate College's own template fixes the
% LAYOUT, and it is the only place the layout is published.
%
% Appendix A, verbatim: "List your degree as it appears on MyUI. If you have an
% official sub-track, you may include it in parentheses after your degree." /
% "List the correct month and year of your graduation." / "List the names of
% your committee members. Your thesis supervisor should be listed first,
% followed by a comma and the phrase 'Thesis Supervisor.'" / "There is no page
% number on the Title Page."
%
% And from the manual's prose: "Only designate your thesis supervisor(s).
% Academic titles and/or degrees earned should not be listed." So every other
% member is a bare name -- pass an EMPTY role to \committeemember for them, and
% "Thesis Supervisor" (or "Dissertation Supervisor") as the role for the first.
%
% From the template: the title is centred and "Either Single or Double Spaced";
% everything down to the date is centred; and the committee block is the one
% LEFT-ALIGNED element on the page, set against a tab stop with the label on the
% same line as the supervisor. That alignment was read from the template's
% paragraph properties, not guessed -- every paragraph above it carries
% w:jc="center" and the five committee paragraphs carry none.
%
% \graddate must hold the month and year of graduation.
\newlength{\iowa@clabel}
\renewcommand{\thesistitlepage}{%
	\loadgeometry{nopagenumber}%
	\begin{titlepage}%
		\thispagestyle{empty}\singlespacing
		\centering
		\vspace*{0.5in}%
		\thesis@tagtitle{Title}{\@title}\par
		\vspace{5\baselineskip}
		by\par
		\vspace{\baselineskip}
		\@author\par
		\vspace{5\baselineskip}
		A \thesis@docnoun{} submitted in partial fulfillment\\
		of the requirements for the \thesis@degree\\
		degree in \thesis@program{} in the\\
		Graduate College of\\
		The University of Iowa\par
		\vspace{2\baselineskip}
		\thesis@date\par
		\vfill
		% The committee block: left-aligned, label and supervisor on one line,
		% remaining members aligned under the supervisor.
		\settowidth{\iowa@clabel}{\thesis@DocNoun{} Committee:~}%
		\noindent\raggedright
		\thesis@DocNoun{} Committee:~%
		\begin{minipage}[t]{\dimexpr\textwidth-\iowa@clabel\relax}
			\raggedright
			% Plain names, one per line: Iowa collects no signatures in the
			% document, so the class's default signature rule is wrong here, and
			% its leading \vskip would push the supervisor off the label's line.
			\renewcommand{\thesis@memberline}[2]{%
				##1\ifx\relax##2\relax\else, ##2\fi\par
			}%
			\thesis@committee
		\end{minipage}\par
		\vspace{0.5in}
	\end{titlepage}%
	\loadgeometry{normal}%
}

% --- Committee approval page --------------------------------------------------
% Iowa's filed document contains none. Appendix B is a closed table of "Required
% and Optional Elements" -- Title Page, Copyright Page, Dedication, Epigraph,
% Acknowledgments, Abstract, Public Abstract, Table of Contents, List of Tables,
% List of Figures, List of Symbols, List of Abbreviations, Preface, Manuscript
% Chapters, Appendices, References/Bibliography -- and no approval or signature
% page appears in it. The committee is named on the title page instead.
%
% NO SIGNATURES ANYWHERE. Approval happens in ProQuest after the Graduate
% College's own format check: "Once I approve your thesis, ProQuest will send an
% email to your committee members, letting them know your thesis is ready for
% approval ... a link that allows your committee members to approve the thesis."
\thesisnoapprovalpage
