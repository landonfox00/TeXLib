% CURRENCY. These are the REVISED requirements: "Students should implement these
% changes beginning in the Spring 2026 semester." The revision removed Roman
% numeral pagination, banned the List of Figures and List of Tables, banned full
% justification, and changed the title page's spacing and capitalization. Any
% older description of an Illinois thesis — including the previous shape of this
% institution's own pages — is superseded.
%
% ============================================================================
% ONE HARD REQUIREMENT THE CLASS VIOLATES BY DEFAULT: JUSTIFICATION.
% ============================================================================
%     "The body text must be left-aligned."
%     "Body text that is fully justified will not be accepted."
%
% LaTeX sets justified text by default, so a texlib-thesis document is out of
% spec at Illinois until the filer says otherwise. There is no profile field for
% it, and setting it here would mean a profile reaching past the filing figures
% into the document's typography — the same line already held at Arizona State
% over its typeface list. The one-line fix belongs in the document preamble:
%
%     \AtBeginDocument{\raggedright}
%
% If the library ever grows a justification field, Illinois is the institution
% that motivates it, and this is the least ambiguous case in the directory —
% unlike a choice among eight fonts, there is exactly one correct action.
%
% TWO THINGS NO PROFILE FIELD CAN EXPRESS.
%   * Pagination. Illinois has NO Roman numerals at all any more: "There will be
%     no longer be Roman numeral page numbers. The abstract will begin on Arabic
%     numeral 2, and the numbering will continue in sequence throughout." The
%     title page carries no number; a copyright page, if present, is also
%     unnumbered and pushes the abstract to 3. A document filed here must not
%     call \frontmatter.
%   * The List of Figures and List of Tables are BANNED: "Students may not
%     include a List of Figures and/or a List of Tables in the document unless
%     they have an exception that they have previously discussed with the Thesis
%     Office" — and the stated reason is accessibility, that such a list
%     "reduces the accessibility and readability of the document". Do not emit
%     one.
%
% THE TYPEFACE RULE IS A BAN, so the class passes it: "Script and ornamental
% fonts will not be accepted", with body text "from 10- to 12-point". The class
% is Latin Modern at 12pt, which is neither banned kind and is inside the range.

\thesisinstitution{University of Illinois Urbana-Champaign}

% "All margins must be a minimum of 1 inch on all sides. No text (other than
% page numbers) may extend into the margin." A MINIMUM with no stated ceiling;
% the profile takes the minimum, which is also what the title-page requirements
% state flatly ("The margins should be 1 inch on all sides of the page"). A
% wider margin is in spec; a narrower one is not.
\thesissetgeometry{left=1in, right=1in, top=1in, bottom=1in}

% "Spacing of the body text may be from 1.5 lines to Double and must remain
% consistent throughout the main text." A CHOICE; the profile takes double,
% which is the class default. \thesissetspacing{onehalf} is equally in spec.
% Single spacing is allowed within the main text for "titles, headings,
% footnotes, endnotes, references, lengthy quotations, bulleted or numbered
% lists, figure or table captions, or material in an appendix".
\thesissetspacing{double}

% --- Accessibility -----------------------------------------------------------
% Illinois has rebuilt its format requirements AROUND accessibility — it is the
% stated reason for the 2026 revision, and accessibility language runs through
% the requirements rather than sitting on a separate page. But it never names a
% WCAG level, a PDF standard, a tagged PDF or a document language, so none of
% the checkable declarations is set. What it does state is recorded below.
%
% AN AMBIGUITY WORTH A REVIEWER'S EYE. The Format Requirements page states alt
% text as a requirement — "must include descriptive alt text" — while the
% Updates page frames the same three items more softly: "We strongly encourage
% students to incorporate these accessibility standards during the Summer 2026
% deposit period." The two pages are the Graduate College's own and they do not
% agree on force. The stronger wording is quoted below, on the page that IS the
% requirements; the softer framing is quoted too rather than dropped.
\thesisaccessibilityrequirement{Alt text}
	{``Figures, photographs, charts, graphs, maps, line drawings, and musical
	 examples must include descriptive alt text. \dots{} Your alt text should not
	 duplicate the caption.'' For \LaTeX, Illinois points at the tagging
	 project's package status list and at the \texttt{alt} key a compatible
	 package offers, illustrated with a tikzpicture whose optional argument
	 carries \texttt{alt=\{Alt text goes here\}}. ``Note that auto-generated alt
	 text is typically general, unhelpful, and often inaccurate.''}
\thesisaccessibilityrequirement{Headings as structure}
	{``When using \LaTeX, all headings should be created using sectioning
	 commands. \dots{} This creates the proper structured outline for assistive
	 technology.'' The title page ``should be created with
	 \texttt{\textbackslash{}maketitle} to properly assign it as the title and
	 should NOT be contained in sectioning.''}
\thesisaccessibilityrequirement{Table headers}
	{``Tables should have a row and/or column header and simple structure.''}
\thesisaccessibilityrequirement{No List of Figures or Tables}
	{Banned, on accessibility grounds: such a list ``is no longer essential and
	 reduces the accessibility and readability of the document''. An exception
	 must be agreed with the Thesis Office in advance.}
\thesisaccessibilityrequirement{No scanned documents}
	{``Scanned documents (such as surveys, IRB letters, etc.) should not be
	 included.''}
\thesisaccessibilityrequirement{Force of these rules}
	{The Format Requirements page states alt text with ``must''; the Thesis
	 Updates page says of alt text, table headers and heading styles that ``We
	 strongly encourage students to incorporate these accessibility standards
	 during the Summer 2026 deposit period''. Both are the Graduate College's
	 own wording. No WCAG level, PDF standard or tagging rule is named anywhere.}

% --- Title page ----------------------------------------------------------------
% Reproduced from the Doctoral Title Page Example, cross-checked against the
% Master's example and the Master's Committee variant. The specimen:
%
%     Coffee Consumption of Graduate Students Trying to Finish Dissertations
%     by
%     Anne E. Garvie
%
%     Dissertation
%     Submitted in partial fulfillment of the requirements
%     for the degree of Doctor of Philosophy in Food Science and Human Nutrition
%     in the Graduate College of the
%     University of Illinois Urbana-Champaign, 2026
%
%     Urbana, Illinois
%
%     Doctoral Committee:
%          Professor Laurence Strongarm, Chair
%          Professor Joseph Green, Director of Research
%          Assistant Professor G.L. Foreman
%          Associate Professor Celia Barerra, Northern Illinois University
%
% "Urbana, Illinois" appears in all three specimens and in NONE of the prose —
% the requirements page's text block runs only as far as the university line.
% It is here because the specimens print it.
%
% The prose rules, each honoured:
%   - "The margins should be 1 inch on all sides of the page."
%   - "There should be no page number anywhere on the page."
%   - "The title should be the only boldface type on the title page."
%   - "The following items should be in title case ...: title, your name, and
%     either Thesis (for master's thesis) or Dissertation (for doctoral
%     dissertations)" — so nothing here is uppercased; the strings are emitted
%     as the document supplies them.
%   - The text block: "Submitted in partial fulfillment of the requirements /
%     for the degree of <earned degree> in <major> / with a concentration in
%     <concentration> / with a minor in <minor> / in the Graduate College of the
%     / University of Illinois Urbana-Champaign, <year of degree conferral>".
%   - "The major must fit entirely onto line 2 of the text block."
%   - "The year in line 6 is the year of degree conferral period for which the
%     student will deposit" — so \graddate takes a bare year.
%   - Committee: "the committee chair should be listed first, and the director of
%     research should be listed second", with ", Chair" and ", Director of
%     Research" appended; faculty listed with their professorial title spelled in
%     full BEFORE the name; and "Affiliations should be listed only for committee
%     members who are not University of Illinois Urbana-Champaign faculty".
%     Pass the whole rendered line as the member name and the role as the second
%     argument: \committeemember{Professor Laurence Strongarm}{Chair}.
%
% No signature rules anywhere: Illinois collects no signatures in the document.
\renewcommand{\thesistitlepage}{%
	\loadgeometry{nopagenumber}%
	\begin{titlepage}%
		\thispagestyle{empty}\singlespacing
		\begin{center}
			\vspace*{0.4in}%
			% The extra braces are load-bearing: "The title should be the only
			% boldface type on the title page", and an unscoped \bfseries here
			% bolds every line down to "Urbana, Illinois".
			\thesis@tagtitle{Title}{{\bfseries\@title}}\par
			\vspace{\baselineskip}
			by\par
			\vspace{\baselineskip}
			\@author\par
			\vspace{\stretch{2}}
			\thesis@DocNoun\par
			\vspace{\baselineskip}
			Submitted in partial fulfillment of the requirements\\
			for the degree of \thesis@degree\ in \thesis@program
			\ifx\uiuc@concentration\@empty\else
				\\with a concentration in \uiuc@concentration
			\fi
			\ifx\uiuc@minor\@empty\else
				\\with a minor in \uiuc@minor
			\fi
			\\in the Graduate College of the\\
			University of Illinois Urbana-Champaign, \thesis@date\par
			\vspace{\baselineskip}
			Urbana, Illinois\par
		\end{center}
		\vspace{\stretch{1}}
		% Left-aligned, at the margin, with the members indented -- as all three
		% specimens print it.
		\noindent
		\ifx\uiuc@cmteheading\@empty
			\def\@tempa{dissertation}%
			\ifx\thesis@docnoun\@tempa Doctoral Committee:\else Adviser:\fi
		\else
			\uiuc@cmteheading
		\fi
		\par
		\vspace{\baselineskip}
		\begingroup
		\leftskip=2.5em\relax
		\renewcommand{\thesis@memberline}[2]{%
			\noindent ##1\ifx\relax##2\relax\else, ##2\fi\par
		}%
		\thesis@committee
		\endgroup
		\vspace*{\stretch{2}}
	\end{titlepage}%
	\loadgeometry{normal}%
}
