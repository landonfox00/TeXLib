% Feature fixture: two or more \sqrt[n]{...} inside ONE formula -- the shape
% that kills a tagged build whenever MathML structure elements are enabled.
%
% luamml 0.9.2 (TeX Live 2026) keeps references to a radical's delimiter and
% degree noads and writes marked-content attributes to them from the
% structure-element writer after mlist_to_hlist has freed them. Two nth-roots in
% one formula is enough for the freed slot to be recycled, and LuaTeX aborts the
% run outright -- "(nodes): trying to delete an attribute reference of a non
% attribute node", no PDF. It reproduces on a bare article + \DocumentMetadata,
% so it is not ours, and only the mathml-SE path reaches that writer.
%
% Two roots in SEPARATE formulas are fine. That is exactly why nothing else in
% the corpus caught this: the full suite ran 36/36 green while the bug was live,
% and the fix that dodges it (ACCESSIBLE_DOCMETA emitting mathml-AF alone) was
% reverted by a branch switch without a single gate going red.
%
% This fixture is tagged accessible, so the accessible gate builds these four
% formulas under whatever ACCESSIBLE_DOCMETA currently says. Restore mathml-SE
% while luamml is still broken and this build dies instead of passing quietly.
%
% The formulas are the radical laws from Math 126's R.6 -- the section where two
% nth-roots in one line is the lesson rather than an accident.
\documentclass{didactic}

\begin{document}

\section{Radical Laws}

RADPRODUCTMARK: $\sqrt[n]{a \cdot b} = \sqrt[n]{a} \cdot \sqrt[n]{b}$

RADQUOTIENTMARK: $\displaystyle \sqrt[n]{\frac{a}{b}} = \frac{\sqrt[n]{a}}{\sqrt[n]{b}}$

RADCANCELMARK: $\sqrt[n]{a^n} = a, \quad \left(\sqrt[n]{a}\right)^n = a$

RADEXPONENTMARK: $a^{\frac{1}{3}} = \sqrt[3]{a}$ and $a^{\frac{1}{n}} = \sqrt[n]{a}$

\end{document}
