% Smoke build for course-metadata.sty's document-facing behaviour: custom-key
% absorption, the derived \Get/\Set commands, the \meta alias, and the PDF
% document title.
%
% Custom keys: each one below must (a) be absorbed by the catch-all without
% erroring and (b) mint a working \Get<Key> getter.  Rendering all three markers
% proves both halves: a regressed catch-all (.unknown) errors under
% -halt-on-error, and a regressed getter derivation leaves \Get... undefined ->
% also a build error.  Asserted by EXPECT_TEXT["Test/Metadata"] in smoke_test.py.
%
% hyperref is loaded because course-metadata sets the PDF title through
% \hypersetup, guarded by \ifdefined and deferred to \AtBeginDocument.  Without
% it this fixture built untitled and was the tree's only PDF/UA-2 8.11.1 failure
% under --accessible -- a permanently red conformance gate for a path no shipped
% class takes, since all nine reach hyperref.  Loaded AFTER course-metadata on
% purpose: that is the ordering the \AtBeginDocument deferral exists to survive,
% so this also covers the deferral.  The resulting title composes through
% \GetCourse, which makes the accessible build's title check a second, metadata-
% level assertion that \SetCourseTitle below really round-tripped.
\documentclass{article}
\usepackage{course-metadata}
\usepackage{hyperref}

% Loud branch: warnings ON -> the auto-create note is a \PackageWarning (.log,
% non-fatal). Must not halt the build.
\MetaWarnCustomKeys
\metasetup{lecture-hall = {Room 7 CMLECTHALLMARK}}

% Quiet branch: same machinery, warning downgraded to \PackageInfo.
\MetaAllowCustomKeys
\metasetup{ta-name = {Pat CMTANAMEMARK}}

% \Set<Field> command-setter: minted alongside every \Get<Field> getter. Setting
% a predefined field this way must round-trip back through its \Get<Field>.
\SetCourseTitle{Calculus SETCMDMARK}

% \meta is now a thin alias of \metasetup; it must still set metadata (here a
% custom key, absorbed by the quiet branch above) and mint its getter.
\meta{exam-room = {Gym METAALIASMARK}}

\begin{document}
Office: \GetOfficeHours\par   % custom key from coursemeta
Hall:   \GetLectureHall\par   % inline custom key, loud branch
TA:     \GetTaName\par        % inline custom key, quiet branch
Title:  \GetCourseTitle\par   % predefined field set via \SetCourseTitle
Room:   \GetExamRoom\par      % custom key set via the \meta alias
\end{document}
