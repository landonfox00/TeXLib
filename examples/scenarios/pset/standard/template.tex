\begin{definition}[Continuity]
	PSETDEFMARK: a function $f$ is continuous at $a$ if
	$\lim_{x \to a} f(x) = f(a)$.
\end{definition}

\begin{problem}[10]
	PSETPLAINMARK: determine whether $f(x) = x^2$ is continuous at $x = 1$.
\end{problem}
